server {
    listen 80;
    listen [::]:80;
    listen 443 ssl;
    listen [::]:443 ssl;

    server_name 202407391.aplicacionesweb2026.com 3.151.202.136;

    root /var/www/html/angular-app;
    index index.html;

    # Certificados (se aplicarán si entras por el dominio)
    ssl_certificate /etc/letsencrypt/live/202407391.aplicacionesweb2026.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/202407391.aplicacionesweb2026.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # Servir Angular
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Proxy para backend
    location /api/ {
        proxy_pass http://localhost:3000/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
    }
}
